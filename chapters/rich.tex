\subsection{Algorithm}

\begin{algorithmic}[1]
  \Require matrix $A$, vector $b$, tolerance $\varepsilon$
  \State $x \gets 0$, \quad $r \gets b$
  \While{$\lVert r \rVert > \varepsilon$}
    \State $x \gets x + \alpha\, r$
    \State $r \gets b - A x$
  \EndWhile
  \State \Return $x$
\end{algorithmic}
